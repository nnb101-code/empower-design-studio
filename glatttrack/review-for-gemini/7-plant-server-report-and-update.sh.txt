#!/bin/bash
# GlattTrack — report to the manufacturer's support center (v10.36). Every 5 minutes (systemd timer
# gt-report.timer). Sends the plant's health (_plant_report: no kashrut data, no names, no codes;
# v10.38: + the problem reports and the daily totals for billing),
# gets back the support center's commands and the current release, and carries them out:
#   restart_standby · reload_tablets · run_nightly_check · update_now
#   a new release → downloaded over https, sha256 checked, installed with update-app.sh (02 + the
#   security self-test first; a release with a server part waits until nobody works).
# Without internet nothing breaks: the plant works on; the support center shows "no report".
. "$(dirname "$0")/gt-lib.sh"
set +e
D="$(cd "$(dirname "$0")" && pwd)"
[ -n "${GT_CENTER_URL:-}" ] && [ -n "${GT_PLANT_CODE:-}" ] && [ -n "${GT_PLANT_SECRET:-}" ] || exit 0
[ "$(gt_self_role)" = "primary" ] || exit 0
# the old anon key (a JWT, eyJ...) also goes as Bearer; a new publishable key (sb_publishable_...) only as apikey
AUTH=(); case "${GT_CENTER_KEY:-}" in eyJ*) AUTH=(-H "Authorization: Bearer $GT_CENTER_KEY");; esac
RPC(){ curl -s -m 20 -X POST -H 'Content-Type: application/json' -H "apikey: $GT_CENTER_KEY" "${AUTH[@]}" \
            "$GT_CENTER_URL/rest/v1/rpc/$1" --data-binary @-; }
J(){ "${GT_PSQL:-psql}" -X -q -t -A -h "$GT_PGHOST" -p "$GT_PG_PORT" -U "${GT_PGUSER:-postgres}" -d "$GT_DB" "$@"; }

body="$(J -v code="$GT_PLANT_CODE" -v sec="$GT_PLANT_SECRET" <<'SQL'
select jsonb_build_object('p_code', :'code', 'p_secret', :'sec', 'p_report', _plant_report());
SQL
)"
resp="$(printf '%s' "$body" | RPC sc_report)"
ok="$(J -v r="$resp" <<'SQL'
select coalesce((nullif(:'r', '')::jsonb ->> 'ok')::boolean, false);
SQL
)"
[ "$ok" = "t" ] || { gt_log "support center: no answer / refused: ${resp:0:200}"; exit 0; }
gt_state centerReportAt "$(date -Is)"
# v10.38: the tickets' status + the center's last reply → the team leader sees them (support_tickets)
J -v r="$resp" >/dev/null <<'SQL'
insert into plant_state (key, value)
select 'supportTickets', coalesce(nullif(:'r', '')::jsonb -> 'tickets', '[]'::jsonb)::text
on conflict (key) do update set value = excluded.value, updated_at = now();
SQL

done_cmd(){ printf '%s' "{\"p_code\":\"$GT_PLANT_CODE\",\"p_secret\":\"$GT_PLANT_SECRET\",\"p_id\":$1,\"p_result\":\"$2\"}" | RPC sc_command_done >/dev/null; }
working(){ [ "$(J -c "select exists (select 1 from animals_pilot where updated_at > now() - interval '10 minutes')")" = "t" ]; }

# the release: install when it is new (the app at once — tablets ask / wait themselves; a server part only when nobody works)
install_release(){   # $1 = force (update_now)
  local rel; rel="$(J -v r="$resp" <<'SQL'
select concat_ws(' ', r ->> 'version', r ->> 'appUrl', r ->> 'appSha256', coalesce(r ->> 'sqlUrl', '-'), coalesce(r ->> 'sqlSha256', '-'))
  from (select nullif(:'r', '')::jsonb -> 'release' as r) x where x.r is not null and x.r <> 'null'::jsonb;
SQL
)"
  [ -n "$rel" ] || return 0
  read -r V AURL ASHA SURL SSHA <<<"$rel"
  local cur; cur="$(J -c "select value from plant_state where key = 'appVersion'")"
  [ "$V" = "$cur" ] && [ "$1" != "force" ] && return 0
  if [ "$SURL" != "-" ] && working && [ "$1" != "force" ]; then gt_log "release $V has a server part — waits until nobody works"; return 0; fi
  local T; T="$(mktemp -d)"
  curl -fsS -m 120 -o "$T/app.html" "$AURL" && [ "$(sha256sum "$T/app.html" | cut -d' ' -f1)" = "$ASHA" ] \
    || { gt_log "release $V: the app file did not download or its sha256 is wrong — not installed"; gt_state lastUpdateError "download/sha256 $V"; rm -rf "$T"; return 1; }
  if [ "$SURL" != "-" ]; then
    curl -fsS -m 120 -o "$T/02.sql" "$SURL" && [ "$(sha256sum "$T/02.sql" | cut -d' ' -f1)" = "$SSHA" ] \
      || { gt_log "release $V: the server file did not download or its sha256 is wrong — not installed"; gt_state lastUpdateError "download/sha256 $V"; rm -rf "$T"; return 1; }
    bash "$D/update-app.sh" "$T/app.html" "$T/02.sql"; rc=$?
  else bash "$D/update-app.sh" "$T/app.html"; rc=$?; fi
  rm -rf "$T"
  [ $rc -eq 0 ] && gt_state_del lastUpdateError || gt_state lastUpdateError "update-app.sh exit $rc ($V)"
  return $rc
}

# the support center's commands (only these four)
while read -r id cmd; do
  [ -n "$id" ] || continue
  case "$cmd" in
    restart_standby)   if [ -n "${GT_STANDBY_RESTART_CMD:-}" ]; then eval "$GT_STANDBY_RESTART_CMD" >/dev/null 2>&1 && done_cmd "$id" "restarted" || done_cmd "$id" "failed"; else done_cmd "$id" "GT_STANDBY_RESTART_CMD not set"; fi ;;
    reload_tablets)    J -c "update settings_pilot set settings = settings || jsonb_build_object('forceReloadAt', now()::text), updated_at = now(), device_id = 'server' where id = 1" >/dev/null && done_cmd "$id" "sent (tablets update when not in use)" ;;
    run_nightly_check) bash "$D/nightly-check.sh" >/dev/null 2>&1; done_cmd "$id" "ran" ;;
    update_now)        install_release force && done_cmd "$id" "installed" || done_cmd "$id" "failed — see the plant log" ;;
  esac
done < <(J -v r="$resp" <<'SQL'
select (c ->> 'id') || ' ' || (c ->> 'cmd') from jsonb_array_elements(coalesce(nullif(:'r', '')::jsonb -> 'commands', '[]'::jsonb)) c;
SQL
)
install_release
exit 0

#!/bin/bash
# GlattTrack — update the plant server (v10.36, owner: no team-leader approval, never harming the work).
#   update-app.sh <new-app.html> [<new-02.sql>]   install a new version
#   update-app.sh --rollback                      back to the version before the last update
# Order: the server first (02, then the security self-test 03 must PASS), and only then the app file
# the tablets load. A tablet that finds the new file updates by itself when nobody works on it, or
# asks "update now / later" (the app does that). If anything fails, nothing changes for the tablets.
. "$(dirname "$0")/gt-lib.sh"
set -u
APP="${GT_APP_FILE:?set GT_APP_FILE in gt-plant.conf — the app file the web server serves}"
PSQL(){ "${GT_PSQL:-psql}" -X -h "$GT_PGHOST" -p "$GT_PG_PORT" -U "${GT_PGUSER:-postgres}" -d "$GT_DB" "$@"; }
say(){ echo "$*"; gt_log "update: $*"; }
ver_of(){ grep -o "var GT_APP_VERSION='[a-z][0-9][^']*';" "$1" 2>/dev/null | head -1 | sed "s/.*='\(.*\)';/\1/"; }

if [ "${1:-}" = "--rollback" ]; then
  [ -f "$APP.prev" ] || { say "nothing to roll back to ($APP.prev missing)"; exit 1; }
  cp -f "$APP.prev" "$APP.tmp" && mv -f "$APP.tmp" "$APP"
  gt_state appVersion "$(ver_of "$APP")"; gt_state appUpdatedAt "$(date -Is)"
  say "rolled back to $(ver_of "$APP") — the tablets go back by themselves"; exit 0
fi

NEW="${1:?usage: update-app.sh <new-app.html> [<new-02.sql>] | --rollback}"; SQL="${2:-}"
[ -f "$NEW" ] || { say "file not found: $NEW"; exit 1; }
NV="$(ver_of "$NEW")"; OV="$(ver_of "$APP")"
[ -n "$NV" ] || { say "not a GlattTrack app file (no version inside): $NEW"; exit 1; }
[ "$(gt_self_role)" = "primary" ] || { say "this is the backup server — update the main server"; exit 1; }

# 1. the server part (re-runnable), then the security self-test must pass
if [ -n "$SQL" ]; then
  [ -f "$SQL" ] || { say "file not found: $SQL"; exit 1; }
  PSQL -v ON_ERROR_STOP=1 -q -f "$SQL" >/dev/null 2>"/tmp/gt-update-02.err" || { say "02 FAILED — nothing changed for the tablets: $(tail -2 /tmp/gt-update-02.err)"; exit 2; }
fi
if [ -n "${GT_SELFTEST_SQL:-}" ] && [ -f "$GT_SELFTEST_SQL" ]; then
  out="$(PSQL -q -f "$GT_SELFTEST_SQL" 2>&1)"
  echo "$out" | grep -q 'PASSED — ' || { say "security self-test did NOT pass — the app was NOT updated (tablets stay on $OV)"; exit 3; }
fi

# 2. the app file: the old one kept for --rollback, the new one put in place in one step
cp -f "$APP" "$APP.prev" 2>/dev/null || true
cp -f "$NEW" "$APP.tmp" && mv -f "$APP.tmp" "$APP" || { say "could not write $APP"; exit 4; }
gt_state appVersion "$NV"; gt_state appUpdatedAt "$(date -Is)"
say "updated $OV → $NV (self-test passed). Tablets update by themselves; rollback: update-app.sh --rollback"

#!/bin/bash
# GlattTrack — sticker printing through the plant server (v10.38). Runs all the time (gt-print-relay.service),
# on the main server only. (1) Sends each new print job (ZPL from a station tablet, print_job_add) to its
# network printer, IP:9100, and marks it sent / failed. (2) When the team leader asks to search
# (printer_scan_request), looks on the plant's own network for label printers (port 9100) and leaves the list
# in plant_state 'printersFound'. Sends only to the plant's own network and a printer port — the same rule
# as the server (_printer_addr_ok).
. "$(dirname "$0")/gt-lib.sh"
set +e
ADDR_RE='^(10\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}|172\.(1[6-9]|2[0-9]|3[01])\.[0-9]{1,3}\.[0-9]{1,3}|192\.168\.[0-9]{1,3}\.[0-9]{1,3}):(9100|9101|9102|6101)$'
[ -n "${GT_PRINT_TEST_ADDR_RE:-}" ] && ADDR_RE="$GT_PRINT_TEST_ADDR_RE"   # tests only (a fake printer on this machine)
send_job(){   # $1 = id, $2 = ip:port
  local id="$1" ip="${2%:*}" port="${2##*:}" f rc err=""
  [[ "$2" =~ $ADDR_RE ]] || { gt_psql -c "update print_jobs set status = 'failed', error = 'address not allowed' where id = $id" >/dev/null; return; }
  f="$(mktemp)"
  gt_psql -c "select data from print_jobs where id = $id" | base64 -d > "$f" 2>/dev/null
  if [ ! -s "$f" ]; then err="empty job"
  else timeout 15 bash -c 'exec 3<>"/dev/tcp/$1/$2" && cat "$3" >&3' _ "$ip" "$port" "$f" 2>/dev/null; rc=$?
       [ $rc -eq 0 ] || err="printer $ip:$port does not answer (code $rc)"; fi
  rm -f "$f"
  if [ -z "$err" ]; then gt_psql -c "update print_jobs set status = 'sent', sent_at = now(), error = null where id = $id" >/dev/null
  else gt_psql -c "update print_jobs set status = 'failed', error = \$e\$$err\$e\$ where id = $id" >/dev/null; gt_log "print job $id: $err"; fi
}
scan(){
  local nets n ip found="" model
  nets="$( { ip -o -4 addr show scope global 2>/dev/null | awk '{print $4}' | cut -d/ -f1; hostname -I 2>/dev/null | tr ' ' '\n'; } | grep -E '^(10\.|172\.(1[6-9]|2[0-9]|3[01])\.|192\.168\.)' | sed 's/\.[0-9]*$//' | sort -u)"
  [ -n "${GT_PRINTER_NETS:-}" ] && nets="$GT_PRINTER_NETS"      # e.g. "192.168.10 192.168.11" (the first three numbers)
  for n in $nets; do
    found="$found $(seq 1 254 | xargs -P 64 -I{} bash -c 'timeout 0.5 bash -c "exec 3<>/dev/tcp/$0.{}/9100" 2>/dev/null && echo "$0.{}"' "$n")"
  done
  local lines=""
  for ip in $found; do
    # a Zebra / ZPL printer answers ~HI with its model (others stay silent — then only the address)
    model="$(timeout 3 bash -c 'exec 3<>"/dev/tcp/$0/9100"; printf "~HI\r\n" >&3; timeout 2 head -c 120 <&3' "$ip" 2>/dev/null | tr -cd '[:alnum:] ._,-' | cut -c1-60)"
    lines="$lines$ip|$model"$'\n'
  done
  # (psql substitutes :'l' only in what it reads, not in -c)
  gt_psql -v l="$lines" >/dev/null <<'SQL'
insert into plant_state (key, value)
select 'printersFound', jsonb_build_object('at', now(), 'printers', coalesce(jsonb_agg(jsonb_build_object('ip', split_part(x, '|', 1), 'port', 9100,
       'model', nullif(split_part(x, '|', 2), '')) order by x), '[]'::jsonb))::text
  from unnest(string_to_array(trim(:'l', E'\n'), E'\n')) x where x <> ''
on conflict (key) do update set value = excluded.value, updated_at = now();
SQL
  gt_log "printer search: $(echo $found | wc -w) found on: $nets"
}
last_beat=0
while true; do
  if [ "$(gt_self_role)" != "primary" ]; then sleep 5; continue; fi
  now=$(date +%s)
  if [ $((now - last_beat)) -ge 30 ]; then gt_state printRelayAt "$(date -Is)" >/dev/null; last_beat=$now; fi
  while read -r id printer; do [ -n "$id" ] && send_job "$id" "$printer"; done \
    < <(gt_psql -c "select id || ' ' || printer from print_jobs where status = 'new' order by id limit 50")
  if [ "$(gt_psql -c "select coalesce((select value::timestamptz from plant_state where key = 'printerScanRequestedAt')
                        > coalesce((select (value::jsonb ->> 'at')::timestamptz from plant_state where key = 'printersFound'), '-infinity'), false)")" = "t" ]; then
    scan
  fi
  sleep "${GT_PRINT_POLL:-0.5}"
done
